# honeypot_receiver.py from flask import Flask, request, jsonify import json import logging from datetime import datetime from pathlib import Path app = Flask(__name__) # Логирование logging.basicConfig( filename='credentials_received.log', level=logging.INFO, format='[%(asctime)s] %(message)s' ) # Папка для сохранения логов LOG_DIR = Path('/var/log/honeypot') LOG_DIR.mkdir(exist_ok=True) @app.route('/api/upload', methods=['POST']) def receive_credentials(): """Получить украденные пароли""" try: # Получить данные data = request.get_json() attacker_ip = request.remote_addr x_forwarded_for = request.headers.get('X-Forwarded-For') if x_forwarded_for: attacker_ip = x_forwarded_for.split(',')[0] # Информация о запросе request_info = { 'timestamp': datetime.now().isoformat(), 'attacker_ip': attacker_ip, 'user_agent': request.headers.get('User-Agent'), 'email_from': data.get('email_from'), 'credentials_count': data.get('count', 0), 'credentials': data.get('credentials', []) } # Логировать в файл log_file = LOG_DIR / f"credentials_{datetime.now().strftime('%Y-%m-%d_%H-%M-%S')}.json" with open(log_file, 'w', encoding='utf-8') as f: json.dump(request_info, f, indent=2, ensure_ascii=False) # Логировать также в основной лог logging.info(f""" IP: {attacker_ip} Email: {data.get('email_from')} Credentials count: {data.get('count', 0)} File saved: {log_file} """) # Вывести в консоль print(f""" ╔═══════════════════════════════════════════╗ ║ 🚨 CREDENTIALS RECEIVED 🚨 ║ ╠═══════════════════════════════════════════╣ ║ IP: {attacker_ip} ║ Email: {data.get('email_from')} ║ Count: {data.get('count', 0)} credentials ║ Time: {datetime.now()} ║ Saved: {log_file} ╚═══════════════════════════════════════════╝ """) # Отправить ответ (malware ожидает ответ от сервера) response = { 'status': 'success', 'message': 'Data received successfully', 'received': True } return jsonify(response), 200 except Exception as e: logging.error(f"Error: {e}") return jsonify({'status': 'error', 'message': str(e)}), 400 @app.route('/health', methods=['GET']) def health_check(): """Проверка что сервер живой""" return jsonify({'status': 'ok'}), 200 if __name__ == '__main__': # Запустить на порту 5000 (потом можно nginx перенаправить на 443) app.run(host='0.0.0.0', port=5000, debug=False)
